import {test} from 'node:test';import assert from 'node:assert/strict';import fs from 'node:fs/promises';import {createHmac,randomUUID} from 'node:crypto';import {PackageSchema,verifyRequest,digest,selectPrompts} from '../src/contracts.mjs';import {reduceProgress} from '../src/repository.mjs';
const pkg=PackageSchema.parse(JSON.parse(await fs.readFile(new URL('../content/pl-basics.json',import.meta.url),'utf8')));
const oldPackage=PackageSchema.parse(JSON.parse(await fs.readFile(new URL('../content/archive/pl-basics-1.0.0.json',import.meta.url),'utf8')));
test('published older lesson packages remain compatible without borrowing new step visuals',()=>{assert.equal(oldPackage.version,'1.0.0');assert.equal(oldPackage.example.steps.find(s=>s.id==='screens').example,undefined);assert.ok(pkg.example.steps.find(s=>s.id==='screens').example);assert.deepEqual(oldPackage.lessons.find(l=>l.id==='role').sections,[]);});
test('published catalog has six lessons, ten simulation steps and valid unique IDs',()=>{assert.equal(pkg.lessons.length,6);assert.equal(pkg.example.steps.length,10);assert.throws(()=>PackageSchema.parse({...pkg,playerVersion:2}));assert.throws(()=>PackageSchema.parse({...pkg,script:'alert(1)'}));});
test('capability filtering and contextual ranking never invent installed operations',()=>{const none=selectPrompts(pkg,{capabilities:[],stage:'welcome'});assert.deepEqual(none.map(x=>x.id),['simple-help']);const design=selectPrompts(pkg,{capabilities:['architecture-design','build-plan','outcome-discovery'],stage:'architecture-design'});assert.ok(['screens','review-design'].includes(design[0].id));assert.equal(design.length,3);});
test('request authority is bound to body, method, route and expiry',()=>{const secret='s'.repeat(48),now=Date.now(),context=Buffer.from(JSON.stringify({tenantId:'test',userId:randomUUID(),capabilities:[],stage:'welcome',aud:'platformtraining',exp:Math.floor(now/1000)+30})).toString('base64url'),raw='{}';const signature=createHmac('sha256',secret).update('POST\n/v1/progress\n'+digest(raw)+'\n'+context).digest('hex');const req={method:'POST',url:'/v1/progress',headers:{'x-training-context':context,'x-training-signature':signature}};assert.equal(verifyRequest(req,raw,secret,now).tenantId,'test');assert.throws(()=>verifyRequest(req,'{"userId":"other"}',secret,now));assert.throws(()=>verifyRequest({...req,url:'/v1/experience'},raw,secret,now));assert.throws(()=>verifyRequest(req,raw,secret,now+90000));});
test('guide advances only after required sample choice and respects pause',()=>{let l={version:0,payload:{mode:'guided',attempt:null}};l=reduceProgress(l,{action:'start'},pkg);l=reduceProgress(l,{action:'next'},pkg);assert.throws(()=>reduceProgress(l,{action:'next'},pkg));l=reduceProgress(l,{action:'choice',choiceId:'recommend'},pkg);l=reduceProgress(l,{action:'pause'},pkg);assert.throws(()=>reduceProgress(l,{action:'next'},pkg));l=reduceProgress(l,{action:'resume'},pkg);l=reduceProgress(l,{action:'next'},pkg);assert.equal(l.payload.attempt.step,2);assert.equal(l.payload.attempt.answers.users,'recommend');});
test('direct mode does not erase or finish the guided attempt',()=>{const a=reduceProgress({version:0,payload:{mode:'guided',attempt:null}},{action:'start'},pkg),b=reduceProgress(a,{action:'mode',mode:'direct'},pkg);assert.deepEqual(a.payload.attempt,b.payload.attempt);assert.equal(b.payload.mode,'direct');});
